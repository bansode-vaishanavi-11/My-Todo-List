async function enableNotifications() { if (!("serviceWorker" in navigator) || !("PushManager" in window) || !("Notification" in window)) { alert("Push Notifications not supported"); return; } try { const permission = await Notification.requestPermission(); if (permission !== "granted") { alert("Notification Permission Denied"); return; } const registration = await navigator.serviceWorker.ready; alert("Notifications Enabled! 🔔"); console.log( "Service Worker Ready:", registration ); } catch (error) { alert("Error: " + error.message); } }
